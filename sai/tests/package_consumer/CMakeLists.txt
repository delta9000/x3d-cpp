cmake_minimum_required(VERSION 3.24)
project(sai_consumer LANGUAGES CXX)

if(SAI_SOURCE_DIR)
    add_subdirectory("${SAI_SOURCE_DIR}" sdk EXCLUDE_FROM_ALL)
else()
    find_package(x3d_sai CONFIG REQUIRED)
endif()
if(TARGET x3d_sai_doctest_main)
    message(FATAL_ERROR "Embedding the SDK must not add its tests")
endif()

add_executable(kernel_consumer author_inspect.cpp)
target_link_libraries(kernel_consumer PRIVATE x3d::sai_reference_kernel)
add_executable(metadata_consumer generated_author_inspect.cpp)
target_link_libraries(metadata_consumer PRIVATE x3d::sai_reference_metadata)

add_executable(provider_consumer provider_reference_test.cpp)
target_link_libraries(provider_consumer PRIVATE
    x3d::sai_reference_provider x3d::sai_provider_fixture)
add_executable(provider_schema_consumer provider_schema_test.cpp)
target_link_libraries(provider_schema_consumer PRIVATE x3d::sai_provider_fixture)
get_target_property(_provider_links x3d::sai_provider INTERFACE_LINK_LIBRARIES)
if(_provider_links MATCHES "sai_(experimental|metadata|reference)")
    message(FATAL_ERROR "Portable provider target links a reference implementation")
endif()

enable_testing()
add_test(NAME provider_consumer COMMAND provider_consumer)
add_test(NAME provider_schema_consumer COMMAND provider_schema_consumer)
add_test(NAME kernel_consumer COMMAND kernel_consumer)
add_test(NAME metadata_consumer COMMAND metadata_consumer)

foreach(kind IN ITEMS kernel metadata)
    add_library(${kind}_shared SHARED shared.cpp)
    set_target_properties(${kind}_shared PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
    if(kind STREQUAL "metadata")
        target_compile_definitions(${kind}_shared PRIVATE SAI_CONSUMER_METADATA)
        target_link_libraries(${kind}_shared PRIVATE x3d::sai_reference_metadata)
    else()
        target_link_libraries(${kind}_shared PRIVATE x3d::sai_reference_kernel)
    endif()
    add_executable(${kind}_shared_consumer shared_main.cpp)
    target_link_libraries(${kind}_shared_consumer PRIVATE ${kind}_shared)
    add_test(NAME ${kind}_shared_consumer COMMAND ${kind}_shared_consumer)
endforeach()

add_executable(provider_scene provider_scene.cpp)
target_link_libraries(provider_scene PRIVATE x3d::sai_reference_provider)
add_test(NAME provider_scene COMMAND provider_scene)
