cmake_minimum_required(VERSION 3.24)
project(x3d_sai_coexistence LANGUAGES CXX C)
include(CTest)

option(SAI_COEXISTENCE_INSTALLED "Consume installed packages instead of source trees" OFF)
if(SAI_COEXISTENCE_INSTALLED)
    find_package(x3d_cpp CONFIG REQUIRED)
    find_package(x3d_sai CONFIG REQUIRED)
else()
    get_filename_component(_default_cpp_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
    set(X3D_CPP_SOURCE_DIR "${_default_cpp_root}" CACHE PATH "x3d-cpp source checkout")
    set(X3D_SAI_SOURCE_DIR "" CACHE PATH "Independent x3d-sai source checkout")
    if(NOT EXISTS "${X3D_SAI_SOURCE_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "Set X3D_SAI_SOURCE_DIR to the independent x3d-sai checkout")
    endif()
    # This consumer owns its test graph; neither package gains a dependency on
    # the other. Avoid optional backends and unrelated examples in this check.
    set(X3D_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(X3D_CPP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(X3D_CPP_BUILD_SCRIPT OFF CACHE BOOL "" FORCE)
    set(X3D_CPP_BUILD_CGLTF OFF CACHE BOOL "" FORCE)
    set(X3D_CPP_ENABLE_DEV_TOOLING OFF CACHE BOOL "" FORCE)
    add_subdirectory("${X3D_CPP_SOURCE_DIR}" x3d-cpp EXCLUDE_FROM_ALL)
    add_subdirectory("${X3D_SAI_SOURCE_DIR}" x3d-sai EXCLUDE_FROM_ALL)
endif()

# Exercise all four combinations: lexical header order and CMake usage/link
# order are independent. Include-guard collisions can otherwise hide an ODR
# violation, while a single lucky include-path order can hide header shadowing.
set(_consumers "")
foreach(_headers IN ITEMS runtime_first sai_first)
    foreach(_links IN ITEMS runtime_first sai_first)
        set(_target "sai_coexistence_${_headers}_${_links}")
        add_executable(${_target} main.cpp)
        target_compile_features(${_target} PRIVATE cxx_std_20)
        if(_headers STREQUAL "sai_first")
            target_compile_definitions(${_target} PRIVATE SAI_HEADERS_FIRST=1)
        endif()
        if(_links STREQUAL "sai_first")
            target_link_libraries(${_target} PRIVATE x3d::sai_metadata x3d_cpp::sdk)
        else()
            target_link_libraries(${_target} PRIVATE x3d_cpp::sdk x3d::sai_metadata)
        endif()
        add_test(NAME ${_target} COMMAND ${_target})
        list(APPEND _consumers ${_target})
    endforeach()
endforeach()
add_custom_target(sai_coexistence_all DEPENDS ${_consumers})
